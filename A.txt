-- ===== ФУНКЦИИ ESP =====
local espObjects = {}

function enableESP()
    for _, p in pairs(Players:GetChildren()) do
        if p ~= player and p.Character then
            createESP(p)
        end
    end
    for _, v in pairs(workspace:GetDescendants()) do
        if v:IsA("Model") and (v.Name:lower():find("drone") or v.Name:lower():find("turret") or v.Name:lower():find("aa") or v.Name:lower():find("zenit")) then
            createESPForObject(v)
        end
    end
end

function disableESP()
    for _, obj in pairs(espObjects) do
        obj:Destroy()
    end
    espObjects = {}
end

function createESP(target)
    if not target.Character then return end
    local head = target.Character:FindFirstChild("Head")
    if not head then return end

    local bill = Instance.new("BillboardGui")
    bill.Name = "ESP_" .. target.Name
    bill.Size = UDim2.new(0, 200, 0, 50)
    bill.Adornee = head
    bill.AlwaysOnTop = true
    bill.MaxDistance = aimbotRange
    bill.Parent = head
    table.insert(espObjects, bill)

    local label = Instance.new("TextLabel", bill)
    label.Size = UDim2.new(1, 0, 1, 0)
    label.BackgroundTransparency = 1
    label.Text = target.Name .. " | " .. string.format("%.0fm", (head.Position - character.HumanoidRootPart.Position).Magnitude)
    label.TextColor3 = Color3.fromRGB(0, 200, 255)
    label.TextScaled = true
    label.Font = Enum.Font.GothamBold
end

function createESPForObject(obj)
    local pos = obj:FindFirstChild("PrimaryPart") or obj:FindFirstChild("Head") or obj:FindFirstChild("HumanoidRootPart")
    if not pos then return end

    local bill = Instance.new("BillboardGui")
    bill.Name = "ESP_" .. obj.Name
    bill.Size = UDim2.new(0, 150, 0, 30)
    bill.Adornee = pos
    bill.AlwaysOnTop = true
    bill.MaxDistance = aimbotRange
    bill.Parent = pos
    table.insert(espObjects, bill)

    local label = Instance.new("TextLabel", bill)
    label.Size = UDim2.new(1, 0, 1, 0)
    label.BackgroundTransparency = 1
    label.Text = obj.Name .. " | " .. string.format("%.0fm", (pos.Position - character.HumanoidRootPart.Position).Magnitude)
    label.TextColor3 = Color3.fromRGB(255, 100, 0)
    label.TextScaled = true
    label.Font = Enum.Font.GothamBold
end

-- ===== АВТО-СПОТИНГ =====
local spottedTargets = {}

function autoSpot()
    for _, p in pairs(Players:GetChildren()) do
        if p ~= player and p.Character and p.Character:FindFirstChild("HumanoidRootPart") then
            local root = p.Character.HumanoidRootPart
            local dist = (root.Position - character.HumanoidRootPart.Position).Magnitude
            if dist <= aimbotRange then
                if not spottedTargets[p] then
                    spottedTargets[p] = true
                    game:GetService("ReplicatedStorage").Events.SpotTarget:FireServer(p)
                end
            else
                spottedTargets[p] = nil
            end
        end
    end
end

-- ===== AIMBOT =====
function getClosestInCircle()
    local closest = nil
    local closestDist = math.huge
    local mouse = player:GetMouse()
    local center = Vector2.new(mouse.X, mouse.Y)

    for _, p in pairs(Players:GetChildren()) do
        if p ~= player and p.Character and p.Character:FindFirstChild("HumanoidRootPart") then
            local root = p.Character.HumanoidRootPart
            local screenPos, onScreen = camera:WorldToScreenPoint(root.Position)
            if onScreen then
                local dist2D = (Vector2.new(screenPos.X, screenPos.Y) - center).Magnitude
                local dist3D = (root.Position - character.HumanoidRootPart.Position).Magnitude
                if dist2D <= aimbotRadius and dist3D <= aimbotRange and dist3D < closestDist then
                    closest = p
                    closestDist = dist3D
                end
            end
        end
    end
    return closest
end
